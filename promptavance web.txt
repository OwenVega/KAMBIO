            }
            catch (Exception ex)
            {

            var resultado =
                await _service.ObtenerDisputaPorId(id);

            if (resultado == null)
                return NotFound();

            return Ok(resultado);
        }

            var resultado =

            if (!resultado)
                return BadRequest();

            return Ok(

        }
